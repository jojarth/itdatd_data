<SpellEntry title="Bless">

    <FeatureSection title="Creature Interactions">
        <FeatureSubsection title="Non-Player Characters">
        * In addition to the normal effects of this spell, a non-player character gains a +1 magic bonus to morale.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Spell Improvements">
        <FeatureSubsection title="When you can cast 3rd-level spells">

        </FeatureSubsection>
        <FeatureSubsection title="When you can cast 5th-level spells">
        * You can target 5 creatures instead of 4.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Components">
        Present your holy symbol toward the targets and speak a brief prayer of blessing.
    </FeatureSection>
</SpellEntry>

    </FeatureSection>

    <FeatureSection title="Spell Improvements">
        <FeatureSubsection title="When you can cast 3rd-level spells">
