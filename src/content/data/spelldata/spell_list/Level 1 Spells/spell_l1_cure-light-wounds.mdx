---
datatype: spell
title: Cure Light Wounds
level: 1
casting: 1 Master Action; Phase 4
range: Touch
duration: Instantaneous
shortDescription: Your touch restores vigor to an ally.
reverseTitle: Inflict Light Wounds
reverseShortDescription: Your touch drains vigor from a foe.
classes:
  - Cleric
  - Theurge
---
import FeatureSection from '../../../../../components/FeatureSection.astro';
import FeatureSubsection from '../../../../../components/FeatureSubsection.astro';
import SpellEntry from '../../../../../components/SpellEntry.astro';

<SpellEntry title="Cure Light Wounds">
    <FeatureSection title="Spell Effect">
        Touch a nearby Valid Creature. The target regains 1d8 + 3 Grit.
    </FeatureSection>

    <FeatureSection title="Creature Interactions">
        <FeatureSubsection title="Valid Creatures">
        * A creature with the **Aeon**, **Animus**, **Cosmic**, or **Fey** keyword is a Valid Creature.
        * A creature with the **Necros** or **Umbra** keyword is a Valid Creature. However, it is entitled to a Saving Throw vs Spell. On a success, nothing happens. On a failure, it suffers 1d8 + 3 points of Sacred damage instead of regaining Grit.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Spell Improvements">
        <FeatureSubsection title="When you can cast 3rd-level spells">
        * Increase the healing provided, as well as the damage to *necros* and *umbra* creatures, to 1d8 + 5.
        </FeatureSubsection>
        <FeatureSubsection title="When you can cast 5th-level spells">
        * Increase the healing provided, as well as the damage to *necros* and *umbra* creatures, to 1d8 + 7.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Components">
        <FeatureSubsection title="Preparation">
        Dab a strip of clean linen with wine, wrap it around one hand, and meditate upon the recovery of flesh and vigor. The linen and wine are consumed when the spell is successfully prepared.
        </FeatureSubsection>

        <FeatureSubsection title="Casting">
        Hold your Casting Hand open before you and utter a prayer or plea of one to five words. Then lay that hand upon the target.
        </FeatureSubsection>
    </FeatureSection>
</SpellEntry>

<SpellEntry title="Inflict Light Wounds" summary="Your touch drains vigor from a foe." variant="reverse" metadata={props.reverseMetadata}>
    <FeatureSection title="Spell Effect">
        Touch a nearby Valid Creature. The target suffers 1d8 + 3 points of Profane damage. The target is entitled to a Save vs Spell. On a success, it takes no damage.
    </FeatureSection>

    <FeatureSection title="Creature Interactions">
        <FeatureSubsection title="Valid Creatures">
        * A creature with the **Aeon**, **Animus**, **Cosmic**, or **Fey** keyword is a Valid Creature.
        * A creature with the **Necros** or **Umbra** keyword is a Valid Creature. Instead of suffering damage, it regains 1d8 + 3 Grit. Such creatures do not require a Saving Throw to gain this benefit. 
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Spell Improvements">
        <FeatureSubsection title="When you can cast 3rd-level spells">
        * Increase the damage dealt, as well as the healing to *necros* and *umbra* creatures, to 1d8 + 5.
        </FeatureSubsection>
        <FeatureSubsection title="When you can cast 5th-level spells">
        * Increase the damage dealt, as well as the healing to *necros* and *umbra* creatures, to 1d8 + 7.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Components">
        <FeatureSubsection title="Preparation">
        Rub grave soil into a strip of burial shroud, wrap it around one hand, and meditate upon the frailty of flesh and vigor. The shroud and soil are consumed when the spell is successfully prepared.
        </FeatureSubsection>

        <FeatureSubsection title="Casting">
        Curl the fingers of your Casting Hand like claws and utter a curse or condemnation of one to five words. Then press that hand against the target.
        </FeatureSubsection>
    </FeatureSection>
</SpellEntry>
