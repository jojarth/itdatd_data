---
datatype: spell
title: Cure Light Wounds
level: 1
casting: 1 Master Action; Phase 4
range: Touch
duration: Instantaneous
classes:
  - Cleric
  - Theurge
shortDescription: Your touch restores vigor to an ally.
---
import FeatureSection from '../../../../../components/FeatureSection.astro';
import SpellEntry from '../../../../../components/SpellEntry.astro';

<SpellEntry title="Cure Light Wounds">
    <FeatureSection title="Spell Effect">
        Touch a nearby Valid Creature. The target regains 1d8 + 3 Grit.
    </FeatureSection>

    <FeatureSection title="Creature Interactions">
        ###### Valid Creatures
        * Creatures with the **Aeon**, **Animus**, **Cosmic**, or **Fey** keywords are Valid Creatures.
        * Creatures with the **Necros** or **Umbra** keywords are Valid Creatures. Instead of recovering Grit, they suffer 1d8 + 3 points of Sacred damage. The target is entitled to a Save vs Spell. On a success, it takes no damage.
    </FeatureSection>

    <FeatureSection title="Spell Improvements">
        ###### When you can cast 3rd-level spells
        * Increase the healing provided, as well as the damage to *necros* and *umbra* creatures, to 1d8 + 5.
        ###### When you can cast 5th-level spells
        * Increase the healing provided, as well as the damage to *necros* and *umbra* creatures, to 1d8 + 7.
    </FeatureSection>

    <FeatureSection title="Components">
        You must utter a brief prayer or plea of one to five words while laying a hand upon the target.
    </FeatureSection>
</SpellEntry>
