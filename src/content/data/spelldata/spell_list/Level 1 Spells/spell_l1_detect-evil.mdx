---
datatype: spell
title: Detect Evil
level: 1
casting: 1 Master Action; Phase 2
range: Self
duration: 10 Minutes per Class Level
shortDescription: You can detect the presence of evil.
reverseTitle: Detect Good
reverseShortDescription: You can detect the presence of pure good.
classes:
  - Cleric
  - Theurge
  - Paladin
---
import FeatureSection from '../../../../../components/FeatureSection.astro';
import FeatureSubsection from '../../../../../components/FeatureSubsection.astro';
import SpellEntry from '../../../../../components/SpellEntry.astro';

<SpellEntry title="Detect Evil">
    <FeatureSection title="Spell Effect">
        While this spell is active, Valid Creatures and Objects appear to glow faintly to you.
    </FeatureSection>

    <FeatureSection title="Creature Interactions">
        <FeatureSubsection title="Valid Creatures">
        * A creature with the **Profane** keyword is a Valid Creature. 
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Object Interactions">
        <FeatureSubsection title="Valid Objects">
        * Objects that are cursed are Valid Objects.
        * Objects affected by the Desecrate spell or similar magic are Valid Objects.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Components">
        <FeatureSubsection title="Preparation">
        Hold a shard of smoked glass before one eye and focus through it while praying or meditating upon hidden corruption. The glass remains intact and can be reused.
        </FeatureSubsection>

        <FeatureSubsection title="Casting">
        Say a short prayer while holding your Casting Hand before your face, palm inward. Pass that hand from one eye to the other, then turn its palm outward.
        </FeatureSubsection>
    </FeatureSection>
</SpellEntry>

<SpellEntry title="Detect Good" summary="You can detect the presence of pure good." variant="reverse" metadata={props.reverseMetadata}>
    <FeatureSection title="Spell Effect">
        While this spell is active, Valid Creatures and Objects appear to glow faintly to you.
    </FeatureSection>

    <FeatureSection title="Creature Interactions">
        <FeatureSubsection title="Valid Creatures">
        * A creature with the **Sacred** keyword is a Valid Creature. 
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Object Interactions">
        <FeatureSubsection title="Valid Objects">
        * Objects affected by the Hallow spell or similar magic are Valid Objects.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Components">
        <FeatureSubsection title="Preparation">
        Wash a shard of clear glass in spring water, then hold it before one eye and focus through it while praying or meditating upon hidden virtue. The glass remains intact and can be reused; the spring water is consumed.
        </FeatureSubsection>

        <FeatureSubsection title="Casting">
        Say a short prayer while holding your Casting Hand before your face, palm inward. Pass that hand from one eye to the other, then turn its palm outward.
        </FeatureSubsection>
    </FeatureSection>
</SpellEntry>
