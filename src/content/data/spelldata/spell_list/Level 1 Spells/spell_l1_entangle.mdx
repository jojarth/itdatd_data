import FeatureSection from '../../../../../components/FeatureSection.astro';

import SpellEntry from '../../../../../components/SpellEntry.astro';
import ConditionLink from '../../../../../components/conditions/ConditionLink.astro';

<SpellEntry title="Entangle">

    <FeatureSection title="Spell Effect">
        Magical, ghostly plants burst in a 20-foot radius from a point that you select in range of the spell. The area is difficult terrain for the spell's duration.

        Each creature inside the area when the spell takes effect makes a Save vs Spell. On a failure, it becomes <ConditionLink id="entangled" />. On a success, it is unaffected, although it must still contend with the difficult terrain. Creatures that enter the area later do not make a Saving Throw.

        **Breaking Free.** An Entangled creature can spend 1 Action to make a Brawn or Agility test. If it succeeds, the Entangled condition ends. The condition also ends when the spell does.
    </FeatureSection>

    <FeatureSection title="Spell Improvements">

        <FeatureSubsection title="When you can cast 3rd-level spells">
            * You may improve the area of effect to 25 feet when you cast the spell.
        </FeatureSubsection>

        <FeatureSubsection title="When you can cast 5th-level spells">
            * You may improve the area of effect to 30 feet when you cast the spell.
        </FeatureSubsection>

    </FeatureSection>

    <FeatureSection title="Components">

        <FeatureSubsection title="Casting">
            Press your Casting Hand toward the ground and curl your fingers upward as though drawing roots from the soil; while speaking a command for the plants to seize.
        </FeatureSubsection>

    </FeatureSection>

</SpellEntry>
