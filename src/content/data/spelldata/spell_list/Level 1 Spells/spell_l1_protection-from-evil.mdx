classes:
  - Cleric

import FeatureSection from '../../../../../components/FeatureSection.astro';
import FeatureSubsection from '../../../../../components/FeatureSubsection.astro';
import SpellEntry from '../../../../../components/SpellEntry.astro';

<SpellEntry title="Protection from Evil">
    <FeatureSection title="Spell Effect">
        For the duration of this spell, Valid Creatures have difficulty harming you.
        
        Valid Creatures cannot physically touch you, preventing them from attacking you with natural weapons. Valid Creatures also take a -2 Magic Penalty to weapon attacks made against you.

        Additionally, you gain a +1 Magic Bonus to Saving Throws against spells and effects generated by Valid Creatures.
    </FeatureSection>

    <FeatureSection title="Creature Interactions">
        <FeatureSubsection title="Valid Creatures">
        * A creature with the **Profane** keyword is a Valid Creature. 
            * When a Valid Creature attempts to attack you or subject you to a spell or effect, it may make a Saving Throw vs Spell if it is three or more levels higher than you or possesses the **Aeon** keyword. On a success, it breaks through your protection for the remainder of the spell. That creature ignores the spell's restrictions and penalties, and you no longer gain the spell's bonus to Saving Throws against spells or effects generated by it.
        </FeatureSubsection>
    </FeatureSection>
</SpellEntry>
