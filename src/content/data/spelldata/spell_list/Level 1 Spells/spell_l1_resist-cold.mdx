import FeatureSubsection from '../../../../../components/FeatureSubsection.astro';

<SpellEntry title="Resist Cold">

    <FeatureSection title="Spell Effect">
        Upon finishing this spell, the creature you touch is warded from cold. They suffer no ill effects for being in extreme cold weather, and gain resistance against Cold damage. 
    </FeatureSection>

    <FeatureSection title="Special Provisions">
        If you cast this spell again, it does not end the previous casting of this spell, allowing you to cast it multiple times.  
    </FeatureSection>

    <FeatureSection title="Components">

        <FeatureSubsection title="Casting">
            Press your Casting Hand to the target's chest and trace a warming sign while speaking a word of shelter.
        </FeatureSubsection>

    </FeatureSection>
