  - Druid

import FeatureSection from '../../../../../components/FeatureSection.astro';
import FeatureSubsection from '../../../../../components/FeatureSubsection.astro';
import SpellEntry from '../../../../../components/SpellEntry.astro';

<SpellEntry title="Soothe">
    <FeatureSection title="Spell Effect">
        Touch a nearby Valid Creature. Beginning on Phase 1 of the following round, the target regains 1d4 + 1 Grit on Phase 1 of each round for the spell's duration.
    </FeatureSection>

    <FeatureSection title="Creature Interactions">
        <FeatureSubsection title="Valid Creatures">
        * A creature with the **Aeon**, **Animus**, **Cosmic**, or **Fey** keyword is a Valid Creature.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Spell Improvements">
        <FeatureSubsection title="When you can cast 3rd-level spells">
        * Increase the duration of this spell to 4 rounds.
        </FeatureSubsection>
        <FeatureSubsection title="When you can cast 5th-level spells">
        * Increase the duration of this spell to 5 rounds.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Components">
        <FeatureSubsection title="Casting">
        Hold your Casting Hand open over the target and utter a prayer or plea of one to five words. Then lay that hand upon the target and slowly draw your thumb across the point of contact.
        </FeatureSubsection>
    </FeatureSection>
</SpellEntry>
