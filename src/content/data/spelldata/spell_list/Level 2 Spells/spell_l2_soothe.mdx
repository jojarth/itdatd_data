casting: 1 Master Action; Phase 4
range: Touch
duration: 3 Rounds
classes:
  - Druid

---
import FeatureSection from '../../../../../components/FeatureSection.astro';
import SpellEntry from '../../../../../components/SpellEntry.astro';

<SpellEntry title="Soothe">
    <FeatureSection title="Spell Effect">
        Touch a nearby Valid Creature. For the duration of this spell, on Phase 1 of each round, the target regains 1d4 + 1 points of Grit. 
    </FeatureSection>
###### Creature Interactions
    * If the target has the **Aeon**, **Animus**, **Cosmic**, or **Fey** keywords, the spell functions normally.
    * If the target has the **Construct, Elemental, Necros** or **Umbra** keywords, the spell has no effect.

    <FeatureSection title="Creature Interactions">
        ###### Valid Creatures
        * Creatures with the **Aeon**, **Animus**, **Cosmic**, or **Fey** keywords are Valid Creatures.
    </FeatureSection>

    <FeatureSection title="Spell Improvements">
        ###### When you can cast 3rd-level spells
        * Increase the duration of this spell to 4 rounds.
        ###### When you can cast 5th-level spells
        * Increase the duration of this spell to 5 rounds.
    </FeatureSection>

    <FeatureSection title="Components">
        You must utter a brief prayer or plea of one to five words while laying a hand upon the target.
    </FeatureSection>
</SpellEntry>
