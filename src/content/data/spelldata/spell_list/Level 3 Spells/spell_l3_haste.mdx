import FeatureSection from '../../../../../components/FeatureSection.astro';
import FeatureSubsection from '../../../../../components/FeatureSubsection.astro';
import EffectTable from '../../../../../components/conditions/EffectTable.astro';
import EffectRow from '../../../../../components/conditions/EffectRow.astro';
import SpellOpposition from '../../../../../components/SpellOpposition.astro';
import SpellEntry from '../../../../../components/SpellEntry.astro';
import ConditionLink from '../../../../../components/conditions/ConditionLink.astro';

<SpellEntry title="Haste">

    <FeatureSection title="Spell Effect">
        Choose one willing creature you can see within range, which can be you. For the duration, it moves and acts with unnatural speed.

        <EffectTable title="Haste Spell Effects">
            <EffectRow title="Quick" href="/playersguide/08-combat/#action-delay">It acts 2 phases earlier than normal, but never before Phase 1.</EffectRow>
            <EffectRow title="Extra Action" href="/playersguide/08-combat/#fewer-actions">It has 1 additional Action each round. This never grants additional Master Actions or Reactions.</EffectRow>
            <EffectRow title="Effortless" href="/playersguide/08-combat/#faltering">Its Master Actions cost 1 fewer Action, to a minimum of 1 Action.</EffectRow>
            <EffectRow title="Fleet">Its speed increases by 10 feet.</EffectRow>
        </EffectTable>

        <FeatureSubsection title="The Cost of Speed">
            Unnatural speed takes its toll. Each time this spell affects a creature, it ages 1 year.

            A body also needs rest between such bursts of speed. If this spell affects a creature again before it has completed a Night's Rest, it also gains 1 level of <ConditionLink id="exhausted" />.
        </FeatureSubsection>

        <SpellOpposition />
    </FeatureSection>

    <FeatureSection title="Components">
        <FeatureSubsection title="Preparation">
            Study the spell while letting a pinch of fine sand run through an hourglass, turning it the moment the last grain falls.
        </FeatureSubsection>

        <FeatureSubsection title="Casting">
            Touch your Casting Hand to your lips, then flick it toward the target as you speak the spell in a single, rapid breath.
        </FeatureSubsection>
    </FeatureSection>

</SpellEntry>

<SpellEntry title="Slow" variant="reverse" metadata={props.reverseMetadata}>

    <FeatureSection title="Spell Effect">
        Choose one creature you can see within range. It must make a Save vs Spell. On a success, the spell has no effect. On a failure, for the duration, it moves and acts as if wading through deep water.

        <EffectTable title="Slow Spell Effects">
            <EffectRow title="Action Delay" href="/playersguide/08-combat/#action-delay">It acts 2 phases later than normal.</EffectRow>
            <EffectRow title="Faltering" href="/playersguide/08-combat/#faltering">Its Master Actions cost 1 additional Action, and its Master Reactions cost 1 Action.</EffectRow>
            <EffectRow title="Reduced Speed" href="/playersguide/08-combat/#reduced-speed">Its speed is reduced by 10 feet.</EffectRow>
        </EffectTable>

        <SpellOpposition reverse />
    </FeatureSection>

    <FeatureSection title="Components">
        <FeatureSubsection title="Preparation">
            Study the spell while slowly pouring a thread of honey from one hand into the other, watching it fall until the last drop lets go.
        </FeatureSubsection>

        <FeatureSubsection title="Casting">
            Point your Casting Hand at the target and draw it slowly back toward your chest, as if pulling against deep water, while speaking the spell in a long, drawn-out breath.
        </FeatureSubsection>
    </FeatureSection>

</SpellEntry>
