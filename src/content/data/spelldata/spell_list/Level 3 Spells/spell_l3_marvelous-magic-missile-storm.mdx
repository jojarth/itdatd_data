import FeatureSection from '../../../../../components/FeatureSection.astro';
import FeatureSubsection from '../../../../../components/FeatureSubsection.astro';
import SpellEntry from '../../../../../components/SpellEntry.astro';

<SpellEntry title="Marvelous Magic Missile Storm">
    <FeatureSection title="Spell Effect">
        To cast this spell, spend 1 Master Action beginning on Phase 4. You conjure 6 magical bolts of force that hover above your head for 1 hour. These bolts cannot be targeted or otherwise interacted with. They may only be launched as described below.

        <FeatureSubsection title="Launching Bolts">
        By spending 1 Action beginning on Phase 2, you may launch any number of your remaining bolts at creatures you can see within the spell's range. If you launch multiple bolts with the same Action, you may divide them among different targets in range, but you must declare all targets before rolling any damage. The bolts strike unerringly and deal 1d4 + 2 force damage. Once a bolt is launched, it is destroyed. 
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Spell Improvements">
        <FeatureSubsection title="When you can cast 6th-level spells">
        * You create 8 bolts instead of 6.
        * Each bolt deals 1d6 + 2 points of force damage.
        </FeatureSubsection>
    </FeatureSection>

    <FeatureSection title="Components">
        <FeatureSubsection title="Preparation">
        Arrange six polished glass beads threaded on copper wire in a circle. Trace the path from bead to bead while studying the spell and visualizing the beads orbiting your head. The beads and wire remain intact and can be reused.
        </FeatureSubsection>

        <FeatureSubsection title="Casting">
        To conjure the bolts, chant the spell aloud while raising your Casting Hand above your head with one finger extended. Trace three widening circles in the air with that finger, then sweep your hand down to your side.

        Whenever you launch one or more bolts, chant the spell again. For each target, sweep your Casting Hand up from your side and finish with your extended finger pointing directly at that target.
        </FeatureSubsection>
    </FeatureSection>
</SpellEntry>
