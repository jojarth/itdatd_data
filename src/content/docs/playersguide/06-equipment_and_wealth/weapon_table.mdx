import { getCollection, render } from "astro:content";

export const properties = await Promise.all(
  (await getCollection("weaponProperties"))
    .sort((a, b) => a.data.title.localeCompare(b.data.title))
    .map(async (entry) => ({
      entry,
      Content: (await render(entry)).Content,
    }))
);

export const features = await Promise.all(
  (await getCollection("weaponFeatures"))
    .sort((a, b) => a.data.title.localeCompare(b.data.title))
    .map(async (entry) => ({
      entry,
      Content: (await render(entry)).Content,
    }))
);

## Weapon Properties

Weapon properties provide a base on how a weapon functions. Weapon properties are applied regardless of your proficiency level with the weapon.
